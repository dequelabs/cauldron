import { LoaderOverlay } from '@deque/cauldron-react';

## Example

It is expected that you choose one of the following:

- Set focus to the loader by using `focusOnInitialRender` or by using `ref` to manually set focus, then after loading is complete shift focus to something logical
- Add `role="alert"` to the `LoaderOverlay`

If you need to block access to the rest of the page the `focusTrap` will prevent keyboard access, while rendering a `Scrim` component will prevent mouse access.

```jsx example
function LoaderOverlayExample() {
  const buttonRef = useRef(null);
  const [loading, setLoading] = useState(false);
  const showLoader = () => {
    setLoading(true);
    setTimeout(() => {
      setLoading(false);
      buttonRef.current?.focus();
    }, 5000);
  };
  return (
    <>
      <Button ref={buttonRef} onClick={showLoader}>
        Show Loader for 5 Seconds
      </Button>
      {loading && (
        <>
          <Scrim show />
          <LoaderOverlay focusTrap focusOnInitialRender>
            <p>
              Explanatory secondary text goes here. Let them know what's
              happening, alright?
            </p>
          </LoaderOverlay>
        </>
      )}
    </>
  );
}
```

### Focus Wrapping

When focus wraps inside a `focusTrap` loader, it skips controls with `tabindex="-1"`, such as the items at the end of this loader's `aria-activedescendant` [ActionList](./ActionList) menu, so Shift+Tab from the cancel button moves to the menu rather than its last item.

```jsx example
function LoaderOverlayFocusWrappingExample() {
  const buttonRef = useRef(null);
  const [loading, setLoading] = useState(false);
  const cancel = () => {
    setLoading(false);
    buttonRef.current?.focus();
  };
  return (
    <>
      <Button ref={buttonRef} onClick={() => setLoading(true)}>
        Generate Report
      </Button>
      {loading && (
        <>
          <Scrim show />
          <LoaderOverlay focusTrap focusOnInitialRender>
            <p>
              Generating your report. You can leave this page, and we'll email
              the report when it's ready.
            </p>
            <Button variant="secondary" onClick={cancel}>
              Cancel
            </Button>
            <ActionList role="menu" aria-label="Go to">
              <ActionListLinkItem href="#dashboard">
                Dashboard
              </ActionListLinkItem>
              <ActionListLinkItem href="#issues">Issues</ActionListLinkItem>
            </ActionList>
          </LoaderOverlay>
        </>
      )}
    </>
  );
}
```
